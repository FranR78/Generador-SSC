function gauge(val,max,tipo){
  var frac=Math.max(0,Math.min(1,val/max)), ang=Math.PI*(1-frac);
  var cx=70,cy=76,r=58;
  var nx=cx+r*0.85*Math.cos(ang), ny=cy-r*0.85*Math.sin(ang);
  var col=tipo==='a'?'var(--alta)':'var(--baja)';
  var arco='M '+(cx-r)+' '+cy+' A '+r+' '+r+' 0 0 1 '+(cx+r)+' '+cy;
  return '<svg viewBox="0 0 140 86">'
    + '<path d="'+arco+'" fill="none" stroke="#dde7ee" stroke-width="10"/>'
    + '<path d="'+arco+'" fill="none" stroke="'+col+'" stroke-width="10" stroke-dasharray="'
    + (Math.PI*r*frac)+' '+(Math.PI*r)+'"/>'
    + '<line x1="'+cx+'" y1="'+cy+'" x2="'+nx.toFixed(1)+'" y2="'+ny.toFixed(1)+'" stroke="#1a1a1a" stroke-width="3"/>'
    + '<circle cx="'+cx+'" cy="'+cy+'" r="5" fill="#1a1a1a"/></svg>';
}

    +   '<div class="gauge b">'+gauge(c.baja,7,'b')+'<div class="val">'+c.baja+' bar</div><div class="lab">Baja</div></div>'
    +   '<div class="gauge a">'+gauge(c.alta,30,'a')+'<div class="val">'+c.alta+' bar</div><div class="lab">Alta</div></div>'
    + '</div>'
    + '<div class="sintoma"><strong>Síntoma:</strong> '+c.sintoma+'</div>'
    + '<h3 class="bloque">1 · Comprobaciones</h3>'
    + bloquesP
    + '<h3 class="bloque">Registro de lo comprobado</h3>'
    + '<ul class="registro" id="reg"></ul>'
    + '<p class="vacio" id="vacio">Aún no has comprobado nada. Diagnosticar sin pruebas es tirar a la diana.</p>'
    + '<h3 class="bloque">2 · Diagnóstico</h3>'
    + '<p class="vacio" style="font-style:normal">Toca el componente averiado en el esquema, o elige una causa:</p>'
    + esquema()
    + '<div class="otras">'+otros+'</div>'
    + '<p class="elegido" id="elegido">Diagnóstico seleccionado: <b>—</b></p>'
    + '<div id="evalua"></div>'
    + '<div class="acciones"><button id="emitir" disabled>Emitir diagnóstico</button>'
    +   '<button id="siguiente" class="sec" disabled>Siguiente ▶</button></div>';

  juego.querySelectorAll('[data-p]').forEach(function(b){
    b.addEventListener('click',function(){ hacerPrueba(b.dataset.p,b); }); });
  juego.querySelectorAll('.comp').forEach(function(el){
    el.addEventListener('click',function(){ elegir(el.dataset.id); }); });
  juego.querySelectorAll('[data-dx]').forEach(function(b){
    b.addEventListener('click',function(){ elegir(b.dataset.dx); }); });
  document.getElementById('emitir').addEventListener('click', emitir);
  document.getElementById('siguiente').addEventListener('click', siguiente);
}

function actualizaMarca(){
  var ok=resultados.filter(function(r){return r.ok;}).length;
  var fu=resultados.filter(function(r){return r.ok&&r.fundamentado;}).length;
  document.getElementById('marca').textContent='Correctos: '+ok+' · Fundamentados: '+fu;
}

function hacerPrueba(id,btn){
  if(cerrado||hechas.indexOf(id)>=0) return;
  hechas.push(id);
  btn.classList.add('hecha'); btn.disabled=true;
  var res=resultado(CASOS[iCaso],id);
  var li=document.createElement('li');
  li.innerHTML='<b>'+NOMBRE_PRUEBA[id]+'</b>'+res.t;
  document.getElementById('reg').appendChild(li);
  document.getElementById('vacio').style.display='none';
}

function elegir(id){
  if(cerrado) return;
  elegido=id;
  juego.querySelectorAll('.comp').forEach(function(el){ el.classList.toggle('sel', el.dataset.id===id); });
  juego.querySelectorAll('[data-dx]').forEach(function(b){ b.classList.toggle('sel', b.dataset.dx===id); });
  document.getElementById('elegido').innerHTML='Diagnóstico seleccionado: <b>'+nombreDx(id)+'</b>';
  document.getElementById('emitir').disabled=false;
}

function emitir(){
  if(cerrado||!elegido) return;
  cerrado=true;
  var c=CASOS[iCaso];
  var ok=c.causa.indexOf(elegido)>=0;
  var faltan=c.clave.filter(function(k){ return hechas.indexOf(k)<0; });
  var fuera=hechas.filter(function(h){ return resultado(c,h).v==='no'; });
  var fundamentado = ok && faltan.length===0;
  resultados.push({ ok:ok, fundamentado:fundamentado, fuera:fuera.length });

  var clase = ok ? (fundamentado?'ok':'medio') : 'mal';
  var titulo = ok ? (fundamentado?'✔ Diagnóstico correcto y fundamentado':'◑ Acertaste, pero sin fundamento') : '✘ Diagnóstico incorrecto';
  var h='<div class="evalua '+clase+'"><h4>'+titulo+'</h4>';
  if(!ok) h+='<p>La causa era: <strong>'+c.causa.map(nombreDx).join(' o ')+'</strong>.</p>';
  if(faltan.length){
    h+='<p>Te faltó comprobar antes de decidir:</p><ul>';
    faltan.forEach(function(k){ h+='<li>'+NOMBRE_PRUEBA[k]+'</li>'; });
    h+='</ul>';
  } else {
    h+='<p>Hiciste las comprobaciones que sostienen el diagnóstico.</p>';
  }
  if(fuera.length){
    h+='<p>Comprobaciones que no procedían en este caso:</p><ul>';
    fuera.forEach(function(k){ h+='<li>'+NOMBRE_PRUEBA[k]+' — '+resultado(c,k).t+'</li>'; });
    h+='</ul>';
  }
  h+='<p>'+c.explica+'</p><p class="ref">Lectura de referencia: '+c.rango+'</p></div>';
  document.getElementById('evalua').innerHTML=h;

  juego.querySelectorAll('[data-p],[data-dx]').forEach(function(b){ b.disabled=true; });
  document.getElementById('emitir').disabled=true;
  document.getElementById('siguiente').disabled=false;
  actualizaMarca();
  document.getElementById('evalua').scrollIntoView({behavior:'smooth',block:'nearest'});
}

function siguiente(){
  iCaso++;
  if(iCaso>=CASOS.length) return fin();
  pintaCaso();
  window.scrollTo({top:0,behavior:'smooth'});
}

function fin(){
  document.getElementById('progreso').style.width='100%';
  document.getElementById('paso').textContent='Resultado final';
  var ok=resultados.filter(function(r){return r.ok;}).length;
  var fu=resultados.filter(function(r){return r.ok&&r.fundamentado;}).length;
  var fuera=resultados.reduce(function(a,r){return a+r.fuera;},0);
  juego.innerHTML='<div class="fin"><div class="nota">'+fu+'/'+CASOS.length+'</div>'
    + '<p>diagnósticos correctos <strong>y fundamentados</strong></p>'
    + '<table><tr><td>Diagnósticos correctos</td><td>'+ok+' de '+CASOS.length+'</td></tr>'
    + '<tr><td>De ellos, con las comprobaciones hechas</td><td>'+fu+'</td></tr>'
    + '<tr><td>Comprobaciones fuera de lugar</td><td>'+fuera+'</td></tr></table>'
    + '<p class="vacio" style="font-style:normal">Acertar sin comprobar no cuenta como diagnóstico: cuenta como suerte.</p>'
    + '<div class="acciones"><button id="reiniciar">Volver a empezar</button></div></div>';
  document.getElementById('reiniciar').addEventListener('click',function(){
    iCaso=0; resultados=[]; pintaCaso(); window.scrollTo({top:0,behavior:'smooth'}); });
}

pintaCaso();
