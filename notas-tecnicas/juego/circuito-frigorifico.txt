/* Componentes del esquema (clicables) + opciones que no son del dibujo */
var COMPONENTES = { compresor:"Compresor", condensador:"Condensador", filtro:"Filtro deshidratador",
                    valvula:"Válvula de expansión", evaporador:"Evaporador" };
var EXTRA = { carga:"Carga de refrigerante", electroventilador:"Electroventilador", correcto:"Todo correcto" };

var iCaso=0, aciertos=0, elegido=null, contestado=false;
var juego = document.getElementById('juego');

function svgGauge(val, max, tipo){
  // semicírculo 180°(izq) → 0°(der); aguja según val/max
  var frac = Math.max(0, Math.min(1, val/max));
  var ang = Math.PI*(1-frac);            // rad
  var cx=75, cy=80, r=62;
  var nx = cx + r*0.86*Math.cos(ang), ny = cy - r*0.86*Math.sin(ang);
  var color = tipo==='a' ? 'var(--alta)' : 'var(--baja)';
  return '<svg viewBox="0 0 150 92">'
    + '<path d="M '+(cx-r)+' '+cy+' A '+r+' '+r+' 0 0 1 '+(cx+r)+' '+cy+'" fill="none" stroke="#dfe7e2" stroke-width="10"/>'
    + '<path d="M '+(cx-r)+' '+cy+' A '+r+' '+r+' 0 0 1 '+(cx+r)+' '+cy+'" fill="none" stroke="'+color+'" stroke-width="10" '
    + 'stroke-dasharray="'+(Math.PI*r*frac)+' '+(Math.PI*r)+'"/>'
    + '<line x1="'+cx+'" y1="'+cy+'" x2="'+nx.toFixed(1)+'" y2="'+ny.toFixed(1)+'" stroke="#222" stroke-width="3"/>'
    + '<circle cx="'+cx+'" cy="'+cy+'" r="5" fill="#222"/></svg>';
}

function esquema(){
  // Lazo cerrado: arriba el lado de alta (compresor→condensador→filtro), la
  // válvula marca la frontera, y abajo el lado de baja (evaporador→compresor).
  var CORTO={compresor:"Compresor",condensador:"Condensador",filtro:"Filtro deshidr.",
             valvula:"Válvula exp.",evaporador:"Evaporador"};
  function box(id,x,y){
    return '<g class="comp" data-id="'+id+'" transform="translate('+x+','+y+')">'
      + '<rect width="110" height="40"></rect>'
      + '<text x="55" y="24">'+CORTO[id]+'</text></g>';
  }
  return '<svg class="circuito" viewBox="0 0 520 185">'
    + '<path class="linea-alta" d="M125 35 H205"/>'
    + '<path class="linea-alta" d="M315 35 H395"/>'
    + '<path class="linea-alta" d="M450 55 V125"/>'
    + '<path class="linea-baja" d="M395 145 H315"/>'
    + '<path class="linea-baja" d="M205 145 H70 V55"/>'
    + '<text class="rot alta-t" x="468" y="95">ALTA</text>'
    + '<text class="rot baja-t" x="52" y="100" text-anchor="end">BAJA</text>'
    + box('compresor',15,15) + box('condensador',205,15) + box('filtro',395,15)
    + box('valvula',395,125) + box('evaporador',205,125)
    + '</svg>'
    + '<div class="leyenda"><span><i class="c-alta"></i>lado de alta presión</span>'
    + '<span><i class="c-baja"></i>lado de baja presión</span></div>';
}

function pintaCaso(){
  contestado=false; elegido=null;
  var c=CASOS[iCaso];
  document.getElementById('paso').textContent='Caso '+(iCaso+1)+' de '+CASOS.length;
  document.getElementById('aciertos').textContent='Aciertos: '+aciertos;
  document.getElementById('progreso').style.width=(iCaso/CASOS.length*100)+'%';

  var ops='';
  Object.keys(COMPONENTES).forEach(function(id){ ops+='<button data-id="'+id+'">'+COMPONENTES[id]+'</button>'; });
  Object.keys(EXTRA).forEach(function(id){ ops+='<button data-id="'+id+'">'+EXTRA[id]+'</button>'; });

  juego.innerHTML =
      '<div class="gauges">'
    +   '<div class="gauge b">'+svgGauge(c.baja,7,'b')+'<div class="val">'+c.baja+' bar</div><div class="lab">Baja</div></div>'
    +   '<div class="gauge a">'+svgGauge(c.alta,30,'a')+'<div class="val">'+c.alta+' bar</div><div class="lab">Alta</div></div>'
    + '</div>'
    + '<div class="sintoma"><strong>Síntoma:</strong> '+c.sintoma+'</div>'
    + esquema()
    + '<div class="pregunta">¿Dónde está la avería? Toca el componente o elige una opción:</div>'
    + '<div class="opciones">'+ops+'</div>'
    + '<div class="feedback" id="fb"></div>'
    + '<div class="acciones"><button id="comprobar" disabled>Comprobar</button>'
    +   '<button id="siguiente" class="sec" disabled>Siguiente ▶</button></div>';

  juego.querySelectorAll('.comp').forEach(function(g){ g.addEventListener('click',function(){ seleccionar(g.dataset.id); }); });
  juego.querySelectorAll('.opciones button').forEach(function(b){ b.addEventListener('click',function(){ seleccionar(b.dataset.id); }); });
  document.getElementById('comprobar').addEventListener('click', comprobar);
  document.getElementById('siguiente').addEventListener('click', siguiente);
}

function seleccionar(id){
  if(contestado) return;
  elegido=id;
  juego.querySelectorAll('.comp').forEach(function(g){ g.classList.toggle('sel', g.dataset.id===id); });
  juego.querySelectorAll('.opciones button').forEach(function(b){ b.classList.toggle('sel', b.dataset.id===id); });
  document.getElementById('comprobar').disabled=false;
}

function nombre(id){ return COMPONENTES[id]||EXTRA[id]||id; }

function comprobar(){
  if(contestado||!elegido) return;
  contestado=true;
  var c=CASOS[iCaso], acierto=c.correcta.indexOf(elegido)>=0;
  if(acierto) aciertos++;
  var fb=document.getElementById('fb');
  fb.className='feedback '+(acierto?'ok':'mal');
  var solucion=c.correcta.map(nombre).join(' o ');
  fb.innerHTML='<h4>'+(acierto?'✔ Correcto':'✘ No es ahí')+'</h4>'
    + (acierto?'':'<p>La avería está en: <strong>'+solucion+'</strong>.</p>')
    + '<p>'+c.explica+'</p><p class="rango">Lectura de referencia: '+c.rango+'</p>';
  juego.querySelectorAll('.opciones button').forEach(function(b){ b.disabled=true; });
  document.getElementById('comprobar').disabled=true;
  document.getElementById('siguiente').disabled=false;
  document.getElementById('aciertos').textContent='Aciertos: '+aciertos;
}

function siguiente(){
  iCaso++;
  if(iCaso>=CASOS.length) return fin();
  pintaCaso();
}

function fin(){
  document.getElementById('progreso').style.width='100%';
  document.getElementById('paso').textContent='Fin';
  var pct=Math.round(aciertos/CASOS.length*100);
  juego.innerHTML='<div class="fin"><div class="nota">'+aciertos+'/'+CASOS.length+'</div>'
    + '<p>Has diagnosticado correctamente el <strong>'+pct+'%</strong> de los casos.</p>'
    + '<div class="acciones"><button id="reiniciar">Volver a empezar</button></div></div>';
  document.getElementById('reiniciar').addEventListener('click',function(){ iCaso=0; aciertos=0; pintaCaso(); });
}

pintaCaso();
